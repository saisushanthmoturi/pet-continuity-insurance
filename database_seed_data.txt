PET CONTINUITY INSURANCE - COMPLETE 24-TABLE MASTER DDL & SEED DATA SCRIPT
================================================================================
Database System: MySQL 8.x / MariaDB
Architecture: 8 Independent Microservice Databases, 24 Tables
Audit Rule: All timestamps (`created_at`, `updated_at`, `submitted_at`, etc.) are 
            autogenerated via `DEFAULT CURRENT_TIMESTAMP` and 
            `ON UPDATE CURRENT_TIMESTAMP`.

================================================================================
CREATE DATABASE IF NOT EXISTS auth_db;
USE auth_db;

-- 1. users
CREATE TABLE IF NOT EXISTS users (
    user_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(100),
    email VARCHAR(150) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(50) NOT NULL,
    status VARCHAR(50) DEFAULT 'ACTIVE',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    last_login_at TIMESTAMP NULL
);

-- 2. refresh_tokens
CREATE TABLE IF NOT EXISTS refresh_tokens (
    token_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT NOT NULL,
    token VARCHAR(500) NOT NULL UNIQUE,
    expires_at TIMESTAMP NOT NULL,
    status VARCHAR(50) DEFAULT 'ACTIVE',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_refreshtoken_user FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);

CREATE DATABASE IF NOT EXISTS underwriting_db;

CREATE TABLE IF NOT EXISTS rating_rules (

    max_value DOUBLE,

    verified_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    next_verification_date DATE,
    remarks TEXT,
    CONSTRAINT fk_pv_ct FOREIGN KEY (caretaker_id) REFERENCES caretakers(caretaker_id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS verification_history (
    history_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    pet_id BIGINT NOT NULL,
    caretaker_id BIGINT NOT NULL,
    verification_type VARCHAR(100) NOT NULL,
    old_status VARCHAR(50),
    new_status VARCHAR(50) NOT NULL,
    verification_method VARCHAR(100),
    verified_by VARCHAR(100),
    verified_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    remarks TEXT,
    CONSTRAINT fk_vh_ct FOREIGN KEY (caretaker_id) REFERENCES caretakers(caretaker_id) ON DELETE CASCADE
);
